```javascript
/**
 * Updates the last activity timestamp for a user.
 * @param {string} handle - User handle
 * @returns {Promise<void>}
 */
async function updateUserActivity(handle) {
    if (!handle) return;
    try {
        const user = await storage.getItem(toKey(handle));
        if (user) {
            const now = Date.now();
            if (!user.lastActivity || now - user.lastActivity > ACTIVITY_UPDATE_INTERVAL) {
                user.lastActivity = now;
                await storage.setItem(toKey(handle), user);
            }
            ONLINE_USERS.add(handle);
        }
    } catch (error) {
        console.error('Failed to update user activity:', handle, error);
    }
}

/**
 * Removes a user from the online users set.
 * @param {string} handle - User handle
 */
function removeOnlineUser(handle) {
    if (!handle) return;
    ONLINE_USERS.delete(handle);
}

/**
 * Clears the user session and removes from online users.
 * @param {import('express').Request} request
 * @param {string} handle
 */
function clearUserSession(request, handle) {
    if (handle) {
        removeOnlineUser(handle);
    }
    if (request.session) {
        request.session.handle = null;
        request.session.csrfToken = null;
        request.session.destroy((err) => {
            if (err) console.error('Failed to destroy session:', err);
        });
    }
}

export function getOnlineUsers() {
    return Array.from(ONLINE_USERS);
}

export function isUserOnline(handle) {
    return ONLINE_USERS.has(handle);
}
```

## 📝 API 使用示例

### 从代码中设置用户过期

```javascript
import storage from 'node-persist';

// 设置用户 24 小时后过期
async function setUserExpiry(handle, hours) {
    const user = await storage.getItem(`user:${handle}`);
    if (user) {
        user.expiry = Date.now() + (hours * 60 * 60 * 1000);
        await storage.setItem(`user:${handle}`, user);
    }
}

// 使用
await setUserExpiry('username', 24);
```

### 清除用户过期时间

```javascript
async function clearExpiry(handle) {
    const user = await storage.getItem(`user:${handle}`);
    if (user) {
        delete user.expiry;
        await storage.setItem(`user:${handle}`, user);
    }
}

await clearExpiry('username');
```

## ⚠️ 注意事项

### 1. 性能优化
- 活动时间每 5 分钟更新一次，避免频繁写入
- 只在 GET 请求时更新，POST 请求不更新

### 2. 时间戳格式
- 所有时间戳都是 UTC 毫秒（`Date.now()`）
- 显示时注意转换为本地时间

### 3. 服务器重启
- `ONLINE_USERS` 是内存中的 Set，重启后清空
- 用户下次访问时会重新添加

### 4. 过期时间精度
- 检查在每个请求时进行
- 精度取决于用户的访问频率

## 🆘 常见问题

### Q: 为什么用户没有被标记为过期？
A: 检查 `user.expiry` 是否设置正确，时间戳应该是未来的时间。

### Q: 活动时间没有更新？
A: 确保用户有 GET 请求，并且距离上次更新超过 5 分钟。

### Q: 如何让用户永不过期？
A: 不设置 `expiry` 字段，或者设置为 `null`/`undefined`。

### Q: 可以在用户过期后自动禁用账户吗？
A: 可以在过期检查时添加：
```javascript
if (user.expiry && user.expiry < Date.now()) {
    user.enabled = false;  // 禁用账户
    await storage.setItem(toKey(handle), user);
    // ... 继续登出逻辑
}
```
